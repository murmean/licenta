#include <GL/glut.h>
#include <math.h>
#include <stdio.h>
#include <stdlib.h>

#define WIDTH 800
#define HEIGHT 600

float cameraX = 10.0f, cameraY = 10.0f, cameraZ = 25.0f;
float lookX = 0.0f, lookY = 0.0f, lookZ = 0.0f;

float robotX = 12.0f, robotY = 0.0f, robotZ = 0.0f;
float robotAngle = 0.0f;
int legAngle = 0;
int legDirection = 1;
int armAngle = 0;

GLfloat light_position[] = {5.0f, 10.0f, 10.0f, 1.0f};
GLfloat light_ambient[] = {0.2f, 0.2f, 0.2f, 1.0f};
GLfloat light_diffuse[] = {0.8f, 0.8f, 0.8f, 1.0f};
GLfloat light_specular[] = {1.0f, 1.0f, 1.0f, 1.0f};

void init() {
    glClearColor(0.25f, 0.4f, 0.5f, 1.0f);
    glEnable(GL_DEPTH_TEST);
    glEnable(GL_LIGHTING);
    glEnable(GL_LIGHT0);
    glEnable(GL_NORMALIZE);
    glEnable(GL_COLOR_MATERIAL);

    glLightfv(GL_LIGHT0, GL_POSITION, light_position);
    glLightfv(GL_LIGHT0, GL_AMBIENT, light_ambient);
    glLightfv(GL_LIGHT0, GL_DIFFUSE, light_diffuse);
    glLightfv(GL_LIGHT0, GL_SPECULAR, light_specular);

    glColorMaterial(GL_FRONT_AND_BACK, GL_AMBIENT_AND_DIFFUSE);
}

void drawTree(float x, float y, float z) {
    // Trunk
    glColor3f(0.4f, 0.2f, 0.1f);
    glPushMatrix();
    glTranslatef(x, y + 0.75f, z);
    glScalef(0.3f, 1.5f, 0.3f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Leaves
    glColor3f(0.0f, 0.5f, 0.0f);
    for (int i = 0; i < 3; i++) {
        glPushMatrix();
        glTranslatef(x, y + 1.5f + i * 0.8f, z);
        glutSolidSphere(0.8f - i * 0.15f, 10, 10);
        glPopMatrix();
    }
}

void drawHills() {
    int hills = 3;
    float hillWidth = 20.0f;
    float hillHeight[] = {3.0f, 2.5f, 4.0f};
    float hillPos[] = {-40.0f, 40.0f, 80.0f};

    glColor3f(0.0f, 0.6f, 0.0f);
    for (int h = 0; h < hills; h++) {
        glPushMatrix();
        glTranslatef(hillPos[h], 0.0f, 0.0f);
        glBegin(GL_TRIANGLE_STRIP);
        for (int i = -20; i <= 20; i++) {
            float x = i * hillWidth / 20.0f;
            float z1 = -20.0f;
            float z2 = 20.0f;
            float height1 = hillHeight[h] * expf(-(x*x)/(hillWidth*hillWidth/4.0f));
            float height2 = height1;
            float dx = - (2.0f * x) / (hillWidth * hillWidth / 4.0f);
            float dy = 1.0f;
            float dz = 0.0f;
            float length = sqrtf(dx*dx + dy*dy + dz*dz);
            dx /= length; dy /= length;
            glNormal3f(dx, dy, dz);
            glVertex3f(x, height1, z1);
            glVertex3f(x, height2, z2);
        }
        glEnd();
        glPopMatrix();
    }
}

void drawTrees() {
    drawTree(-4.0f, 0.0f, -5.0f);
    drawTree(10.0f, 0.0f, -8.0f);
    drawTree(-12.0f, 0.0f, 5.0f);
    drawTree(15.0f, 0.0f, 0.0f);
    drawTree(-5.0f, 0.0f, 15.0f);
}

void drawFence() {
    glColor3f(0.5f, 0.3f, 0.1f);
    float radius = 8.0f;
    int posts = 20;

    for (int i = 0; i < posts; i++) {
        float angle = 2 * M_PI * i / posts;
        float x = radius * cosf(angle);
        float z = radius * sinf(angle) - 1.0f;

        glPushMatrix();
        glTranslatef(x, 0.75f, z);
        glScalef(0.2f, 1.5f, 0.2f);
        glutSolidCube(1.0f);
        glPopMatrix();

        if (i > 0) {
            float prevAngle = 2 * M_PI * (i-1) / posts;
            float prevX = radius * cosf(prevAngle);
            float prevZ = radius * sinf(prevAngle) - 1.0f;

            glBegin(GL_QUADS);
            glNormal3f(0.0f, 1.0f, 0.0f);
            glVertex3f(prevX, 1.5f, prevZ);
            glVertex3f(x, 1.5f, z);
            glVertex3f(x, 1.7f, z);
            glVertex3f(prevX, 1.7f, prevZ);
            glEnd();
        }
    }
}

void drawHouse() {
    // House main structure
    glColor3f(0.8f, 0.6f, 0.4f);
    glPushMatrix();
    glTranslatef(0.0f, 1.5f, -2.0f);

    // Base
    glPushMatrix();
    glScalef(5.0f, 3.0f, 4.0f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Roof
    glColor3f(0.5f, 0.2f, 0.1f);
    glPushMatrix();
    glTranslatef(0.0f, 1.5f, 0.0f);
    glRotatef(-90.0f, 1.0f, 0.0f, 0.0f);
    glutSolidCone(3.5f, 2.0f, 4, 1);
    glPopMatrix();

    // Door
    glColor3f(0.4f, 0.2f, 0.0f);
    glPushMatrix();
    glTranslatef(0.0f, -1.0f, 2.01f);
    glScalef(1.0f, 2.0f, 0.1f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Windows
    glColor3f(0.7f, 0.8f, 1.0f);
    for (int i = -1; i <= 1; i += 2) {
        for (int j = 0; j < 1; j++) {
            glPushMatrix();
            glTranslatef(i * 1.5f, j * 1.0f, 2.01f);
            glScalef(0.7f, 0.7f, 0.1f);
            glutSolidCube(1.0f);
            glPopMatrix();
        }
    }

    glPopMatrix();
}

void drawRobot() {
    // Body
    glColor3f(0.0f, 0.0f, 1.0f);
    glPushMatrix();
    glTranslatef(robotX, robotY + 1.5f, robotZ);
    glScalef(1.5f, 2.0f, 1.0f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Head
    glColor3f(0.8f, 0.8f, 0.8f);
    glPushMatrix();
    glTranslatef(robotX, robotY + 2.7f, robotZ);
    glutSolidSphere(0.5f, 20, 20);

    // Eyes
    glColor3f(1.0f, 0.0f, 0.0f);
    glPushMatrix();
    glTranslatef(0.2f, 0.1f, 0.4f);
    glutSolidSphere(0.1f, 10, 10);
    glPopMatrix();

    glPushMatrix();
    glTranslatef(-0.2f, 0.1f, 0.4f);
    glutSolidSphere(0.1f, 10, 10);
    glPopMatrix();
    glPopMatrix();

    // Arms
    glColor3f(0.1f, 0.1f, 0.5f);
    // Left arm
    glPushMatrix();
    glTranslatef(robotX - 1.2f, robotY + 1.5f, robotZ);
    glRotatef((float)armAngle, 1.0f, 0.0f, 0.0f);
    glScalef(0.3f, 1.5f, 0.3f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Right arm
    glPushMatrix();
    glTranslatef(robotX + 1.2f, robotY + 1.5f, robotZ);
    glRotatef(-(float)armAngle, 1.0f, 0.0f, 0.0f);
    glScalef(0.3f, 1.5f, 0.3f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Legs
    glColor3f(0.1f, 0.1f, 0.5f);
    // Left leg
    glPushMatrix();
    glTranslatef(robotX - 0.5f, robotY + 0.5f, robotZ);
    glRotatef(-(float)legAngle, 1.0f, 0.0f, 0.0f);
    glScalef(0.4f, 1.2f, 0.4f);
    glutSolidCube(1.0f);
    glPopMatrix();

    // Right leg
    glPushMatrix();
    glTranslatef(robotX + 0.5f, robotY + 0.5f, robotZ);
    glRotatef((float)legAngle, 1.0f, 0.0f, 0.0f);
    glScalef(0.4f, 1.2f, 0.4f);
    glutSolidCube(1.0f);
    glPopMatrix();
}

void display() {
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
    glLoadIdentity();

    gluLookAt(cameraX, cameraY, cameraZ,
              lookX, lookY, lookZ,
              0.0f, 1.0f, 0.0f);

    // Ground
    glColor3f(0.15f, 0.45f, 0.15f);
    glBegin(GL_QUADS);
    glNormal3f(0.0f, 1.0f, 0.0f);
    glVertex3f(-50.0f, 0.0f, -50.0f);
    glVertex3f(-50.0f, 0.0f,  50.0f);
    glVertex3f( 50.0f, 0.0f,  50.0f);
    glVertex3f( 50.0f, 0.0f, -50.0f);
    glEnd();

    drawHills();
    drawHouse();
    drawFence();
    drawTrees();
    drawRobot();

    glutSwapBuffers();
}

// The rest of the functions (idle, keyboard, reshape, main) remain unchanged
void idle() {
    legAngle += legDirection * 2;
    armAngle += legDirection * 2;

    if (legAngle > 30) legDirection = -1;
    if (legAngle < -30) legDirection = 1;

    robotZ -= 0.05f;
    if (robotZ < -20.0f) robotZ = 10.0f;

    glutPostRedisplay();
}

void keyboard(unsigned char key, int x, int y) {
    switch(key) {
        case 27:
            exit(0);
            break;
        case 'w':
            cameraZ -= 1.0f;
            break;
        case 's':
            cameraZ += 1.0f;
            break;
        case 'a':
            cameraX -= 1.0f;
            break;
        case 'd':
            cameraX += 1.0f;
            break;
        case 'q':
            cameraY += 1.0f;
            break;
        case 'e':
            cameraY -= 1.0f;
            break;
        case 'i':
            lookY += 0.5f;
            break;
        case 'k':
            lookY -= 0.5f;
            break;
        case 'j':
            lookX -= 0.5f;
            break;
        case 'l':
            lookX += 0.5f;
            break;
    }
    glutPostRedisplay();
}

void reshape(int w, int h) {
    if (h == 0) h = 1;
    float ratio = (float)w / (float)h;

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluPerspective(45.0f, ratio, 1.0f, 100.0f);
    glMatrixMode(GL_MODELVIEW);
}

int main(int argc, char **argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_DEPTH | GLUT_DOUBLE | GLUT_RGB);
    glutInitWindowSize(WIDTH, HEIGHT);
    glutCreateWindow("3D World with Animated Robot");

    init();
    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutKeyboardFunc(keyboard);
    glutIdleFunc(idle);

    glutMainLoop();
    return 0;
}
